#include <iostream>
#include <climits>
#include <algorithm>
using namespace std;

int main()
{
    int n, amount;

    cout << "Enter number of coins: ";
    cin >> n;

    int coins[n];

    cout << "Enter coin denominations: ";
    for (int i = 0; i < n; i++)
    {
        cin >> coins[i];
    }

    cout << "Enter amount: ";
    cin >> amount;

    int dp[amount + 1];

    // Initialize DP array
    for (int i = 0; i <= amount; i++)
    {
        dp[i] = INT_MAX;
    }

    dp[0] = 0;

    // Dynamic Programming
    for (int i = 1; i <= amount; i++)
    {
        for (int j = 0; j < n; j++)
        {
            if (coins[j] <= i &&
                dp[i - coins[j]] != INT_MAX)
            {
                dp[i] = min(
                    dp[i],
                    1 + dp[i - coins[j]]
                );
            }
        }
    }

    if (dp[amount] == INT_MAX)
    {
        cout << "Amount cannot be formed";
    }
    else
    {
        cout << "Minimum coins required = "
             << dp[amount];
    }

    return 0;
}
